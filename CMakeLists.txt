CMAKE_MINIMUM_REQUIRED(VERSION 3.13)
PROJECT(
  ethc
  LANGUAGES C
  VERSION 1.1.2)

INCLUDE(GNUInstallDirs)

IF(NOT CMAKE_BUILD_TYPE)
  SET(CMAKE_BUILD_TYPE Debug)
ENDIF()

MESSAGE(STATUS "ethc: build type: ${CMAKE_BUILD_TYPE}")

set(CMAKE_POSITION_INDEPENDENT_CODE TRUE)
SET(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${PROJECT_SOURCE_DIR}/cmake")

SET(ETHC_SOURCE_FILES
    src/abi.c
    src/ens.c
    src/account.c
    src/address.c
    src/ecdsa.c
    src/bloom.c
    src/internals.c
    src/unit.c
    src/hex.c
    src/rlp.c
    src/mp.c
    src/keccak256.c)

SET(BUILD_SHARED_LIBS ON)

OPTION(ETHC_BUILD_TESTS "Build tests" OFF)
OPTION(ETHC_DISABLE_SHARED_LIBS "Disable shared libs" OFF)
OPTION(ETHC_BUILD_DOCS "Build docs" OFF)
OPTION(ETHC_LIBTOMMATH_STATIC "Statically link libtommath" OFF)
OPTION(ETHC_SECP256K1_STATIC "Statically link libsecp256k1" OFF)

IF(ETHC_DISABLE_SHARED_LIBS)
  SET(BUILD_SHARED_LIBS OFF)
ENDIF()

ADD_LIBRARY(${PROJECT_NAME} ${ETHC_SOURCE_FILES})

SET_TARGET_PROPERTIES(
  ${PROJECT_NAME} PROPERTIES C_STANDARD 99 SOVERSION ${PROJECT_VERSION_MAJOR})

TARGET_COMPILE_OPTIONS(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-sometimes-uninitialized -Wno-sign-compare)

IF(ETHC_TOMMATH_STATIC)
  SET(TOMMATH_INCLUDE_DIRS, libtommath)
  SET(TOMMATH_LINK_LIBRARIES libtommath)

  ADD_SUBDIRECTORY(libtommath)
ELSE()
  FIND_PACKAGE(tommath REQUIRED)
ENDIF()

IF(ETHC_SECP256K1_STATIC)
  SET(SECP256K1_ENABLE_MODULE_RECOVERY ON)
  SET(SECP256K1_INCLUDE_DIRS secp256k1)
  SET(SECP256K1_LINK_LIBRARIES secp256k1)
  SET(SECP256K1_BUILD_BENCHMARK OFF)
  SET(SECP256K1_BUILD_TESTS OFF)
  SET(SECP256K1_BUILD_EXHAUSTIVE_TESTS OFF)
  SET(SECP256K1_BUILD_CTIME_TESTS OFF)
  SET(SECP256K1_BUILD_EXAMPLES OFF)

  ADD_SUBDIRECTORY(secp256k1)
ELSE()
  FIND_PACKAGE(secp256k1 REQUIRED)
ENDIF()

TARGET_LINK_LIBRARIES(${PROJECT_NAME} keccak m
  ${SECP256K1_LINK_LIBRARIES} ${TOMMATH_LINK_LIBRARIES})

TARGET_INCLUDE_DIRECTORIES(
  ${PROJECT_NAME} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
  ${SECP256K1_INCLUDE_DIRS} ${TOMMATH_INCLUDE_DIRS})

IF(ETHC_BUILD_TESTS)
  ADD_CUSTOM_TARGET(libtap.a ALL
    COMMAND make libtap.a
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/libtap)

  ENABLE_TESTING()
  ADD_SUBDIRECTORY(test)
ENDIF()

IF(ETHC_BUILD_DOCS)
  ADD_SUBDIRECTORY(docs)
ENDIF()

MESSAGE(STATUS "ethc: SECP256K1_INCLUDE_DIRS: ${SECP256K1_INCLUDE_DIRS}")
MESSAGE(STATUS "ethc: SECP256K1_LINK_LIBRARIES: ${SECP256K1_LINK_LIBRARIES}")

ADD_SUBDIRECTORY(libkeccak)

INSTALL(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
INSTALL(
  DIRECTORY ${CMAKE_SOURCE_DIR}/include
  DESTINATION .
  FILES_MATCHING
  PATTERN "*.h")

IF(UNIX OR APPLE OR CYGWIN)
  MESSAGE(STATUS "ethc: generating libethc.pc")
  CONFIGURE_FILE(libethc.pc.in libethc.pc @ONLY)

  INSTALL(FILES ${CMAKE_BINARY_DIR}/libethc.pc
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
ENDIF()
